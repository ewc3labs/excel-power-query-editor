// Compiled at runtime by excel-live-sync.ps1 via Add-Type. Kept as a separate file so it is
// readable and reviewable rather than buried in a here-string.
//
// WHY THIS EXISTS AT ALL. The obvious ways to reach an open workbook are both wrong:
//
//   Marshal.GetActiveObject("Excel.Application")
//       Returns ONE instance - whichever registered first. A user with two Excel instances (and
//       Excel does start a second one for some file types, or via /x) can have the workbook in the
//       other one, and we would report "not open" while they stare at it.
//
//   Marshal.BindToMoniker(path)
//       Binds by path across all instances, which sounds perfect. Measured 2026-08-14: on a file
//       that EXISTS BUT IS CLOSED it OPENS THE FILE. Silently opening somebody's workbook because
//       we were checking whether it was open is the worst kind of side effect.
//
// So: enumerate the Running Object Table, and only bind to a moniker that is ALREADY THERE.
// Excel registers each open workbook in the ROT under its full path, so presence in the table is
// exactly the question "is this workbook open, in any instance?" - and GetObject on an existing
// entry cannot start anything.

using System;
using System.Collections.Generic;
using System.Runtime.InteropServices;
using System.Runtime.InteropServices.ComTypes;

public static class RunningObjects
{
    [DllImport("ole32.dll")]
    private static extern int GetRunningObjectTable(int reserved, out IRunningObjectTable prot);

    [DllImport("ole32.dll")]
    private static extern int CreateBindCtx(int reserved, out IBindCtx ppbc);

    /// <summary>Display names of everything currently registered. Excel workbooks appear as full paths.</summary>
    public static string[] Names()
    {
        var names = new List<string>();
        IRunningObjectTable rot;
        IBindCtx ctx;
        if (GetRunningObjectTable(0, out rot) != 0 || rot == null) { return names.ToArray(); }
        if (CreateBindCtx(0, out ctx) != 0 || ctx == null) { return names.ToArray(); }

        IEnumMoniker monikers;
        rot.EnumRunning(out monikers);
        monikers.Reset();

        var slot = new IMoniker[1];
        while (monikers.Next(1, slot, IntPtr.Zero) == 0)
        {
            string name = null;
            try { slot[0].GetDisplayName(ctx, null, out name); }
            catch { continue; }
            if (!string.IsNullOrEmpty(name)) { names.Add(name); }
        }
        return names.ToArray();
    }

    /// <summary>
    /// The object registered under this exact display name, or null.
    ///
    /// Never starts anything: it walks the table and returns what is already registered. If the
    /// name is absent, the answer is null and nothing happens - which is the whole point.
    /// </summary>
    public static object Get(string displayName)
    {
        IRunningObjectTable rot;
        IBindCtx ctx;
        if (GetRunningObjectTable(0, out rot) != 0 || rot == null) { return null; }
        if (CreateBindCtx(0, out ctx) != 0 || ctx == null) { return null; }

        IEnumMoniker monikers;
        rot.EnumRunning(out monikers);
        monikers.Reset();

        var slot = new IMoniker[1];
        while (monikers.Next(1, slot, IntPtr.Zero) == 0)
        {
            string name = null;
            try { slot[0].GetDisplayName(ctx, null, out name); }
            catch { continue; }
            if (name == null) { continue; }
            if (!string.Equals(name, displayName, StringComparison.OrdinalIgnoreCase)) { continue; }

            object found = null;
            try { rot.GetObject(slot[0], out found); }
            catch { return null; }
            return found;
        }
        return null;
    }
}

// A WORKBOOK ON A MAPPED NETWORK DRIVE IS REGISTERED UNDER ITS UNC PATH, NOT ITS DRIVE LETTER.
//
// Reported from a real network share, 2026-09-14. The user opened P:\IT\MedAR_Coder\X.xlsx.
// Excel's own Workbook.FullName said P:\IT\MedAR_Coder\X.xlsx - and the Running Object Table held
// \\medarms01\public\IT\MedAR_Coder\X.xlsx. Same file, two names, and live sync compared the one
// the user sees against the one COM registered, found nothing, and blamed elevation.
//
// Note what nearly hid it: checking FullName appeared to rule the UNC form OUT. FullName is what
// Excel reports; the ROT display name is what COM registered. Only the second is what we match.
//
// This is NOT a heuristic and does not break the no-scoring rule in the helper. WNetGetConnection
// asks Windows what the drive letter is mapped to - the same mapping Excel resolved when it opened
// the file - so the UNC form is the same identity, exactly, not a similar-looking one.
public static class NetworkPaths
{
    [DllImport("mpr.dll", CharSet = CharSet.Unicode)]
    private static extern int WNetGetConnection(string localName, System.Text.StringBuilder remoteName, ref int length);

    /// <summary>
    /// The UNC form of a path on a mapped network drive, or null when it is not on one.
    ///
    /// Null for local drives, SUBST drives, UNC paths already, and anything unparseable - every
    /// case where there is no second name to try.
    /// </summary>
    public static string ToUnc(string path)
    {
        if (string.IsNullOrEmpty(path) || path.Length < 2 || path[1] != ':') { return null; }

        int length = 1024;
        var remote = new System.Text.StringBuilder(length);
        if (WNetGetConnection(path.Substring(0, 2), remote, ref length) != 0) { return null; }

        string root = remote.ToString().TrimEnd('\\');
        if (root.Length == 0) { return null; }
        return root + path.Substring(2);
    }
}
